1. Executive Summary & Architecture

The Inventory Management System provides end-to-end tracking of commercial inventory items, product pricing, stock movement, and supplier logistics. The system ensures complete data integrity through relational primary and foreign key constraints, supports multi-type transactions (Purchases, Sales, Returns), and features a real-time web dashboard connected to a live cloud MySQL database.

3. SQL Schema Definition (DDL)

-- Drop tables in reverse dependency order
DROP TABLE IF EXISTS InventoryTransaction;
DROP TABLE IF EXISTS Product;
DROP TABLE IF EXISTS Supplier;

-- 1. Supplier Table
CREATE TABLE Supplier (
    SupplierID   INT          PRIMARY KEY AUTO_INCREMENT,
    SupplierName VARCHAR(150) NOT NULL,
    ContactName  VARCHAR(100),
    ContactTitle VARCHAR(100),
    Address      VARCHAR(200),
    City         VARCHAR(100),
    Region       VARCHAR(100),
    PostalCode   VARCHAR(20),
    Country      VARCHAR(100),
    Phone        VARCHAR(30),
    Email        VARCHAR(150)
);

-- 2. Product Table
CREATE TABLE Product (
    ProductID       INT            PRIMARY KEY AUTO_INCREMENT,
    ProductName     VARCHAR(150)   NOT NULL,
    Description     TEXT,
    SupplierID      INT,
    UnitPrice       DECIMAL(10,2)  NOT NULL DEFAULT 0.00,
    QuantityInStock INT            NOT NULL DEFAULT 0,
    CONSTRAINT fk_product_supplier
        FOREIGN KEY (SupplierID) REFERENCES Supplier(SupplierID)
        ON UPDATE CASCADE ON DELETE SET NULL
);

-- 3. Inventory Transaction Table
CREATE TABLE InventoryTransaction (
    TransactionID   INT  PRIMARY KEY AUTO_INCREMENT,
    ProductID       INT  NOT NULL,
    TransactionType ENUM('Purchase', 'Sale', 'Return') NOT NULL,
    Quantity        INT  NOT NULL,
    TransactionDate DATE NOT NULL,
    Notes           TEXT,
    CONSTRAINT fk_transaction_product
        FOREIGN KEY (ProductID) REFERENCES Product(ProductID)
        ON UPDATE CASCADE ON DELETE CASCADE
);

Query 2: Stock Movement Summary per Product

SELECT 
    p.ProductName,
    COALESCE(SUM(CASE WHEN t.TransactionType = 'Purchase' THEN t.Quantity ELSE 0 END), 0) AS Purchased,
    COALESCE(SUM(CASE WHEN t.TransactionType = 'Sale'     THEN t.Quantity ELSE 0 END), 0) AS Sold,
    COALESCE(SUM(CASE WHEN t.TransactionType = 'Return'   THEN t.Quantity ELSE 0 END), 0) AS Returned,
    p.QuantityInStock AS CurrentStock
FROM Product p
LEFT JOIN InventoryTransaction t ON p.ProductID = t.ProductID
GROUP BY p.ProductID, p.ProductName, p.QuantityInStock
ORDER BY p.ProductName;

Query 4: Overall Warehouse Valuation

SELECT 
    COUNT(ProductID) AS TotalProducts,
    SUM(QuantityInStock) AS TotalUnitsInStock,
    ROUND(SUM(UnitPrice * QuantityInStock), 2) AS TotalInventoryValue
